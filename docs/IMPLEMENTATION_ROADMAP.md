```sql
CREATE TABLE treatment (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  farm_id UUID NOT NULL REFERENCES farm(id) ON DELETE CASCADE,
  system_id UUID NOT NULL REFERENCES system(id) ON DELETE CASCADE,
  treatment_date DATE NOT NULL,
  treatment_time TIME,
  
  -- Diagnosis
  reason ENUM('disease', 'prevention', 'water_quality', 'parasite', 'vaccination', 'other') NOT NULL,
  disease_diagnosis TEXT,
  
  -- Treatment
  treatment_type ENUM('antibiotic', 'antifungal', 'antiparasitic', 'probiotic', 'vaccination', 'water_quality_adjustment', 'other') NOT NULL,
  product_name TEXT NOT NULL,
  active_ingredient TEXT NOT NULL,
  
  -- Dosing
  dose_amount DECIMAL(10,2) NOT NULL,
  dose_unit ENUM('g', 'ml', 'mg_per_kg', 'ppm', 'IU_per_kg') NOT NULL,
  start_date DATE NOT NULL,
  duration_days INTEGER NOT NULL,
  
  -- Regulatory
  lot_number TEXT,
  expiration_date DATE,
  withdrawal_period_days INTEGER,
  
  -- Response
  effectiveness ENUM('excellent', 'good', 'fair', 'poor') DEFAULT 'unknown',
  mortality_during_treatment DECIMAL(5,2),
  estimated_recovery_pct DECIMAL(5,2),
  
  notes TEXT,
  created_at TIMESTAMP DEFAULT now(),
  updated_at TIMESTAMP DEFAULT now(),
  created_by UUID NOT NULL REFERENCES auth.users(id),
  
  CONSTRAINT valid_dates CHECK (start_date <= treatment_date),
  CONSTRAINT valid_dosage CHECK (dose_amount > 0)
);

CREATE INDEX idx_treatment_farm ON treatment(farm_id);
CREATE INDEX idx_treatment_system ON treatment(system_id);
CREATE INDEX idx_treatment_date ON treatment(treatment_date);
CREATE INDEX idx_withdrawal_period ON treatment(start_date, withdrawal_period_days);

-- Add RLS policies
ALTER TABLE treatment ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view treatments in their farm"
  ON treatment FOR SELECT
  USING (farm_id IN (SELECT farm_id FROM farm_user WHERE user_id = auth.uid()));

CREATE POLICY "Farm managers can create treatments"
  ON treatment FOR INSERT
  WITH CHECK (
    farm_id IN (SELECT farm_id FROM farm_user 
                WHERE user_id = auth.uid() 
                AND role IN ('admin', 'farm_manager'))
  );
```

```typescript
'use client';

import { usePathname, useSearchParams } from 'next/navigation';
import { Breadcrumbs, Link, Typography } from '@mui/material';
import { ChevronRight } from 'lucide-react';

export function PageBreadcrumbs() {
  const pathname = usePathname();
  const searchParams = useSearchParams();

  // Parse URL to build breadcrumb path
  const segments = pathname.split('/').filter(Boolean);
  
  const breadcrumbs = [
    { label: 'Home', href: '/dashboard' },
    ...segments.map((segment, i) => ({
      label: formatSegment(segment),
      href: '/' + segments.slice(0, i + 1).join('/'),
    })),
  ];

  return (
    <Breadcrumbs separator={<ChevronRight size={16} />} className="mb-4">
      {breadcrumbs.map((crumb, i) => 
        i === breadcrumbs.length - 1 ? (
          <Typography key={crumb.href} color="textPrimary">
            {crumb.label}
          </Typography>
        ) : (
          <Link key={crumb.href} href={crumb.href} color="inherit">
            {crumb.label}
          </Link>
        )
      )}
    </Breadcrumbs>
  );
}

function formatSegment(segment: string): string {
  return segment
    .replace(/-/g, ' ')
    .split(' ')
    .map(word => word.charAt(0).toUpperCase() + word.slice(1))
    .join(' ');
}
```

```typescript
'use client';

import { Card, Box, Typography } from '@mui/material';
import { TrendingUp, TrendingDown, Minus } from 'lucide-react';

export function KPICard({ 
  title, 
  value, 
  unit,
  trend, // positive = improvement
  invertTrend = false, // for "bad when high" metrics
  threshold,
  status // 'optimal' | 'caution' | 'critical'
}: KPICardProps) {
  
  const isImprovement = invertTrend ? trend < 0 : trend > 0;
  const trendColor = isImprovement ? '#10b981' : trend === 0 ? '#6b7280' : '#ef4444';

  return (

      <Typography variant="subtitle2" color="textSecondary" className="mb-2">
        {title}
        {threshold && (

          {formatValue(value, unit)}

              variant="body2" 
              style={{ color: trendColor }}
              className="font-semibold"
            >
              {isImprovement ? '+' : ''}{Math.abs(trend).toFixed(1)}%
            </Typography>
          </Box>
        )}
      </Box>
      
      {/* Status badge */}
      <div className={`inline-block px-2 py-1 rounded text-xs font-medium ${getStatusBackground(status)}`}>
        {status === 'optimal' ? '✓ On Target' : status === 'caution' ? '⚠ Caution' : '✗ Critical'}
      </div>
      
      {/* Helpful context */}
      {status === 'caution' && (
        <Typography variant="caption" className="block mt-2 text-amber-700">
          Approaching threshold. Monitor closely.
        </Typography>
      )}
      {status === 'critical' && (
        <Typography variant="caption" className="block mt-2 text-red-700">
          Action recommended. See details →
        </Typography>
      )}
    </Card>
  );
}

function getStatusBorder(status: string): string {
  switch(status) {
    case 'optimal': return 'border-green-500';
    case 'caution': return 'border-amber-500';
    case 'critical': return 'border-red-500';
    default: return 'border-gray-300';
  }
}

function getStatusBackground(status: string): string {
  switch(status) {
    case 'optimal': return 'bg-green-100 text-green-800';
    case 'caution': return 'bg-amber-100 text-amber-800';
    case 'critical': return 'bg-red-100 text-red-800';
    default: return 'bg-gray-100 text-gray-800';
  }
}
```

```typescript
import { Card, TextField, Button, Alert, Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui';
import { fetchLatestSensorReading, type SensorDevice } from '@/lib/sensors/sensor-client';

export function WaterQualityFormEnhanced() {
  const [sensors, setSensors] = useState<SensorDevice[]>([]);
  const [selectedSensor, setSelectedSensor] = useState<string | null>(null);
  const form = useForm();

  // Load from sensor API
  async function loadSensorReading() {
    if (!selectedSensor) return;
    
    try {
      const reading = await fetchLatestSensorReading(selectedSensor);
      form.setValue('temperature', reading.temperature);
      form.setValue('dissolved_oxygen', reading.dissolved_oxygen);
      form.setValue('pH', reading.pH);
      form.setValue('ammonia', reading.ammonia);
      form.setValue('nitrite', reading.nitrite);
      form.setValue('nitrate', reading.nitrate);
      form.setValue('salinity', reading.salinity);
    } catch (error) {
      // Fall back to manual entry
    }
  }

  return (
    <form className="space-y-6">
      <Tabs defaultValue="manual">
        <TabsList>
          <TabsTrigger value="sensor">Sensor Integration</TabsTrigger>
          <TabsTrigger value="manual">Manual Entry</TabsTrigger>
        </TabsList>
        
        <TabsContent value="sensor" className="space-y-4">
          <Alert level="info">
            Connected sensor data auto-fills form. Override individual values as needed.
          </Alert>
          
          <select onChange={(e) => setSelectedSensor(e.target.value)}>
            <option value="">Select Sensor...</option>
            {sensors.map(sensor => (
              <option key={sensor.id} value={sensor.id}>
                {sensor.name} ({sensor.type})
              </option>
            ))}
          </select>
          
          {selectedSensor && (
            <>
              <Button 
                variant="outline"
                onClick={loadSensorReading}
              >
                📡 Load Latest Reading
              </Button>
              
              {sensors.find(s => s.id === selectedSensor)?.calibration_due && (
                <Alert level="warning">
                  Sensor calibration due {sensors.find(s => s.id === selectedSensor)?.calibration_due?.toLocaleDateString()}
                </Alert>
              )}
            </>
          )}
        </TabsContent>
        
        <TabsContent value="manual">
          {/* Manual entry fields with validation */}
          <ParameterInput
            label="Temperature"
            unit="°C"
            bounds={{ min: 0, max: 35 }}
            {...form.register('temperature', { valueAsNumber: true })}
          />
          
          <ParameterInput
            label="Dissolved Oxygen"
            unit="mg/L"
            bounds={{ min: 0, max: 14 }}
            {...form.register('dissolved_oxygen', { valueAsNumber: true })}
          />
          
          {/* ... other parameters ... */}
        </TabsContent>
      </Tabs>
      
      {/* QA/QC Validation */}
      <Card>
        <Alert level="warning" hidden={!hasQAIssue(form.getValues())}>
          {getQAWarning(form.getValues())}
        </Alert>
      </Card>
    </form>
  );
}

function hasQAIssue(values: any): boolean {
  // Bounds checking
  if (values.dissolved_oxygen < 0 || values.dissolved_oxygen > 14) return true;
  if (values.temperature < 0 || values.temperature > 35) return true;
  if (values.pH < 5.5 || values.pH > 8.5) return true;
  return false;
}

function getQAWarning(values: any): string {
  if (values.dissolved_oxygen > 14) return 'DO impossible (max 14 mg/L). Check sensor calibration.';
  if (values.temperature > 35) return 'Temperature extremely high. Verify sensor/entry.';
  if (values.pH < 5.5 || values.pH > 8.5) return 'pH outside normal range for aquaculture. Investigate.';
  return '';
}
```

### 2.3 Add Water Quality Correlation Heatmap

**Create chart component:**

```typescript
// src/components/charts/water-quality-heatmap.tsx

'use client';

import { useMemo } from 'react';
import { Card } from '@mui/material';

export function WaterQualityHeatmap({
  systemId,
  parameter,
  period = 'last_60_days'
}: {
  systemId: string;
  parameter: 'ammonia' | 'temperature' | 'do' | 'ph';
  period?: string;
}) {
  const data = useQuery({
    queryKey: ['water-quality-heatmap', systemId, parameter, period],
    queryFn: () => fetchWQHeatmapData(systemId, parameter, period)
  });

  const heatmapData = useMemo(() => {
    if (!data.data) return [];
    
    // Group measurements by week and system
    const grouped = data.data.reduce((acc, measurement) => {
      const week = getWeekNumber(new Date(measurement.timestamp));
      const key = `${measurement.system_id}-${week}`;
      
      if (!acc[key]) {
        acc[key] = {
          system_id: measurement.system_id,
          week,
          values: []
        };
      }
      
      acc[key].values.push(measurement.value);
      return acc;
    }, {});
    
    // Calculate averages and color intensity
    return Object.values(grouped).map((item: any) => ({
      ...item,
      average: item.values.reduce((a: number, b: number) => a + b, 0) / item.values.length,
      color: getHeatmapColor(item.values)
    }));
  }, [data.data]);

  return (
    <Card className="p-6">
      <h3 className="font-semibold mb-4">{parameter.toUpperCase()} Heatmap (Last 60 Days)</h3>
      
      <div className="grid grid-cols-auto gap-1">
        {/* Week headers */}
        <div />
        {[...Array(8)].map((_, i) => (
          <div key={i} className="text-center text-xs font-medium">Week {i + 1}</div>
        ))}
        
        {/* System rows */}
        {heatmapData
          .filter((item, i, arr) => arr.findIndex(a => a.system_id === item.system_id) === i) // unique systems
          .map(item => (
            <div key={item.system_id} className="contents">
              <div className="text-sm font-medium pr-2">{item.system_id}</div>
              {[...Array(8)].map((_, week) => {
                const weekData = heatmapData.find(
                  d => d.system_id === item.system_id && d.week === week
                );
                return (
                  <div
                    key={`${item.system_id}-${week}`}
                    className="w-8 h-8 rounded border border-gray-300"
                    style={{
                      backgroundColor: weekData?.color || '#f0f0f0'
                    }}
                    title={weekData ? `Avg: ${weekData.average.toFixed(1)}` : 'No data'}
                  />
                );
              })}
            </div>
          ))}
      </div>
      
      <div className="mt-4 flex gap-4 text-sm">
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded" style={{ backgroundColor: '#10b981' }} />
          <span>Optimal</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded" style={{ backgroundColor: '#f59e0b' }} />
          <span>Caution</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-4 h-4 rounded" style={{ backgroundColor: '#ef4444' }} />
          <span>Critical</span>
        </div>
      </div>
    </Card>
  );
}

function getHeatmapColor(values: number[]): string {
  const avg = values.reduce((a, b) => a + b, 0) / values.length;
  
  // This is parameter-specific, would need to be configurable
  if (avg >= 95) return '#10b981';  // Green - optimal
  if (avg >= 70) return '#f59e0b';  // Amber - caution
  return '#ef4444'; // Red - critical
}
```
